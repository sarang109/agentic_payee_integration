rule (modulo E) Register_key:
   [ Fr( ~ltk ) ]
  --[ KeyOnce( $A ) ]->
   [ !Ltk( $A, ~ltk ), !Pk( $A, pk(~ltk) ), Out( pk(~ltk) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Reveal_key:
   [ !Ltk( A, ltk ) ] --[ Reveal( A ) ]-> [ Out( ltk ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Brand_owned:
   [ ] --[ BrandOnce( $b ), OwnsBrand( $E, $b ) ]-> [ !Owns( $b, $E ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Open_account:
   [ Fr( ~p ) ]
  --[ Opens( $O, $E, ~p ) ]->
   [ !AccountOf( $O, $E, ~p ), Out( ~p ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) DV_issue_ID:
   [ !Owns( b, E ), !Ltk( 'DV', kdv ) ]
  --[ IssuedID( b, E ) ]->
   [ Out( sign(<'ID', b, E>, kdv) ) ]

  /* has exactly the trivial AC variant */

lemma executable:
  exists-trace
  "∃ V b p c n #i.
    (Release( V, b, p, c, n ) @ #i) ∧ (¬(∃ A #r. Reveal( A ) @ #r))"
/*
guarded formula characterizing all satisfying traces:
"∃ V b p c n #i.
  (Release( V, b, p, c, n ) @ #i) ∧ ∀ A #r. (Reveal( A ) @ #r) ⇒ ⊥"
*/
simplify
solve( VState( V, b, c, n ) ▶₀ #i )
  case Verifier_start
  solve( !Pk( 'DV', pk(x) ) ▶₁ #i )
    case Register_key
    solve( !Pk( E, pk(x) ) ▶₂ #i )
      case Register_key
      solve( !Pk( O, pk(x) ) ▶₃ #i )
        case Register_key
        solve( !KU( sign(<'ID', $b, $A>, ~ltk) ) @ #vk.7 )
          case DV_issue_ID
          solve( !KU( sign(<'ACC', 'ID', $b, $A>, ~ltk.1) ) @ #vk.9 )
            case Entity_accept_ID
            solve( !KU( sign(<'AG', $A, p, $A.1>, ~ltk.1) ) @ #vk.11 )
              case Entity_authorize_AG
              solve( !KU( sign(<'BETA', ~cart, ~p, ~n>, ~ltk.2) ) @ #vk.14 )
                case Operator_checkout
                solve( !KU( ~p ) @ #vk.11 )
                  case Open_account
                  solve( !KU( ~cart ) @ #vk.18 )
                    case Verifier_start
                    solve( !KU( ~n ) @ #vk.19 )
                      case Verifier_start
                      SOLVED // trace found
                    qed
                  qed
                qed
              qed
            qed
          qed
        qed
      qed
    qed
  qed
