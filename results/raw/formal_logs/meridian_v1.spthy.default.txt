rule (modulo E) Register_key:
   [ Fr( ~ltk ) ]
  --[ KeyOnce( $A ) ]->
   [ !Ltk( $A, ~ltk ), !Pk( $A, pk(~ltk) ), Out( pk(~ltk) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Reveal_key:
   [ !Ltk( A, ltk ) ] --[ Reveal( A ) ]-> [ Out( ltk ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Brand_owned:
   [ ] --[ BrandOnce( $b ), OwnsBrand( $E, $b ) ]-> [ !Owns( $b, $E ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Open_account:
   [ Fr( ~p ) ]
  --[ Opens( $O, $E, ~p ) ]->
   [ !AccountOf( $O, $E, ~p ), Out( ~p ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) DV_issue_ID:

  --[ IssuedID( b, E ) ]->

  /* has exactly the trivial AC variant */
