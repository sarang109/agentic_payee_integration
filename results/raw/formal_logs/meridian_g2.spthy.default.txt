rule (modulo E) Register_key:
   [ Fr( ~ltk ) ]
  --[ KeyOnce( $A ) ]->
   [ !Ltk( $A, ~ltk ), !Pk( $A, pk(~ltk) ), Out( pk(~ltk) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Reveal_key:
   [ !Ltk( A, ltk ) ] --[ Reveal( A ) ]-> [ Out( ltk ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Brand_owned:
   [ ] --[ BrandOnce( $b ), OwnsBrand( $E, $b ) ]-> [ !Owns( $b, $E ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Open_custodian:
   [ Fr( ~h ) ]
  --[ OpensCustodian( $H, $E, ~h ) ]->
   [ !Custodian( $H, $E, ~h ), Out( ~h ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Open_bank_account:
   [ Fr( ~T ) ]
  --[ HoldsAccount( $B, $E, ~T ) ]->
   [ !Account( $B, $E, ~T ), Out( ~T ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) DV_issue_ID:
   [ !Owns( b, E ), !Ltk( 'DV', kdv ) ]
  --[ IssuedID( b, E ) ]->
   [ Out( sign(<'ID', b, E>, kdv) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Entity_authorize_AG:
   [ !Custodian( H, E, h ), !Ltk( E, ke ) ]
  --[ AuthAG( E, h, H ) ]->
   [ Out( sign(<'AG', E, h, H>, ke) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Bank_terminal_binding:
   [ !Account( B, E, T ), !Ltk( B, kb ) ]
  --[ BankBinds( B, E, T ) ]->
   [ Out( sign(<'TERM', E, T>, kb) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Custodian_commit:
   [

  --[ Committed( H, pid, h, T ), CommitOnce( H, pid ) ]->
   [ Out( sign(<'COMMIT', pid, h, T>, kh) ), Pending( H, pid, h, T ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Custodian_transfer_honest:
   [ Pending( H, pid, h, T ), !Ltk( H, kh ) ]
  --[ Transferred( H, pid, h, T ) ]->
   [ Out( sign(<'XFER', pid, h, T>, kh) ) ]

  /* has exactly the trivial AC variant */

rule (modulo E) Custodian_transfer_deviate:
   [ Pending( H, pid, h, T ), !Ltk( H, kh ), In( to ) ]
  --[ Transferred( H, pid, h, to ), Deviates( H, pid ), Neq( to, T ) ]->
   [ Out( sign(<'XFER', pid, h, to>, kh) ) ]

  /* has exactly the trivial AC variant */

lemma executable_g2:
  exists-trace
  "∃ V b pid E h T #i.
    (AcceptG2( V, b, pid, E, h, T ) @ #i) ∧ (¬(∃ A #r. Reveal( A ) @ #r))"
/*
guarded formula characterizing all satisfying traces:
"∃ V b pid E h T #i.
  (AcceptG2( V, b, pid, E, h, T ) @ #i) ∧ ∀ A #r. (Reveal( A ) @ #r) ⇒ ⊥"
*/
simplify
solve( VState( V, b, pid ) ▶₀ #i )
  case Verifier_start
  solve( !Pk( 'DV', pk(x) ) ▶₁ #i )
    case Register_key
    solve( !Pk( E, pk(x) ) ▶₂ #i )
      case Register_key
      solve( !Pk( H, pk(x) ) ▶₃ #i )
        case Register_key
        solve( !Pk( B, pk(x) ) ▶₄ #i )
          case Register_key
          solve( !KU( sign(<'ID', $b, $A>, ~ltk) ) @ #vk.11 )
            case DV_issue_ID
            solve( !KU( sign(<'AG', $A, h, $A.1>, ~ltk.1) ) @ #vk.13 )
              case Entity_authorize_AG
              solve( !KU( sign(<'COMMIT', ~pid, ~h, T>, ~ltk.2) ) @ #vk.15 )
                case Custodian_commit
                solve( !KU( sign(<'TERM', $A, ~T>, ~ltk.3) ) @ #vk.16 )
                  case Bank_terminal_binding
                  solve( !KU( ~h ) @ #vk.12 )
                    case Open_custodian
                    solve( !KU( ~pid ) @ #vk.19 )
                      case Verifier_start
                      solve( !KU( ~T ) @ #vk.16 )
                        case Open_bank_account
                        SOLVED // trace found
                      qed
                    qed
                  qed
                qed
              qed
            qed
          qed
        qed
      qed
    qed
  qed
qed
